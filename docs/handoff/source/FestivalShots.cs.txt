    public static bool Begin()
    {
        foreach (var arg in OS.GetCmdlineUserArgs())
        {
            if (arg.StartsWith("--festival-shots=", StringComparison.Ordinal))
            {
                var shots = new FestivalShots { Name = "FestivalShots", ProcessMode = ProcessModeEnum.Always, ProcessPriority = 100, _directory = arg["--festival-shots=".Length..] };

            }
        }

        return false;
    }

    public override void _Process(double delta)
    {
        _frame++;
        if (_hospital is null)
        {
            if (GetTree().CurrentScene is HospitalScene h && h.IsNodeReady())
            {
                _hospital = h;
                h.Hud.Visible = false;
                if (h.GetNodeOrNull<CanvasItem>("TileCursor") is { } cursor)
                {
                    cursor.Visible = false;
                }

                foreach (var child in h.GetChildren())
                {
                    if (child is CanvasItem ci && child is not Camera2D)
                    {
                        ci.Visible = false;
                    }
                }

                h.Session.Clock.SetSpeed(SimulationSpeed.Paused);
                foreach (var child in h.View3D.GetChildren())
                {
                    if (child is Node3D n && child is not Camera3D && child is not Light3D)
                    {
                        n.Visible = false;
                    }
                }

                _site = new Node3D { Name = "Festival" };
                h.View3D.AddChild(_site);
                Build();
                RenderingServer.FramePreDraw += ApplyMood;
                _frame = 0;
            }

            return;
        }

        if (_frame % 8 != 0)
        {
            return;
        }

        var only = OS.GetCmdlineUserArgs().FirstOrDefault(a => a.StartsWith("--only=", StringComparison.Ordinal))?[7..];
        var views = Views.Where(v => only is null || only.Split(',').Any(o => v.Name.StartsWith(o, StringComparison.Ordinal))).ToArray();
        var index = _shot / 2;
        if (index >= views.Length)
        {
            GetTree().Quit(0);
            return;
        }

        var view = views[index];
        if (_shot % 2 == 0)
        {
            _mood = view.Mood;
            _wet.Visible = _mood == "rain";
            _dry.Visible = _mood != "rain";
            _medRoof.Visible = _mood != "medical";
            _nightOnly.Visible = _mood is "night" or "dusk";
            _density.Visible = _mood == "density";
            _lineup.Rotation = new Vector3(0, _mood == "day2" ? Mathf.Pi : 0, 0);
            var hour = _mood switch { "night" => 23, "dusk" => 20, _ => 14 };
            var clock = _hospital.Session.Clock;
            clock.RestoreState(new ClockState(new DateTime(2027, 7, 3, hour, 20, 0), clock.TickCount, SimulationSpeed.Paused));
            foreach (var p in _crowd)
            {
                p.Visible = true;
            }

            _hospital.Camera.SnapTo(IsoDraw.ToScreen(view.X, view.Y), view.Zoom);
        }
        else
        {
            var image = GetViewport().GetTexture().GetImage();
            DirAccess.MakeDirRecursiveAbsolute(_directory);
            image.SavePng($"{_directory}/{view.Name}.png");
            GD.Print($"[festival] Saved {view.Name}");
        }

        _shot++;
    }

    private void ApplyMood()
    {
        if (_hospital is null || !IsInstanceValid(_hospital))
        {
            return;
        }

        var view = _hospital!.View3D;
        var env = view.GetChildren().OfType<WorldEnvironment>().First().Environment;
        var lights = view.GetChildren().OfType<DirectionalLight3D>().ToList();
        foreach (var l in lights)
        {
            l.DirectionalShadowMaxDistance = 260;
        }

        var sun = lights.OrderByDescending(l => l.LightEnergy).First();
        env.GlowEnabled = _mood is "night" or "dusk";
        env.GlowIntensity = 0.7f;
        env.GlowBloom = _mood == "night" ? 0.02f : 0.08f;
        env.GlowHdrThreshold = 1.0f;
        switch (_mood)
        {
            case "night":
                sun.LightEnergy = 0.1f;
                sun.LightColor = C("#7F96D8");
                env.AmbientLightEnergy = 0.2f;
                foreach (var l in lights.Where(l => l != sun))
                {
                    l.LightEnergy = 0;
                }

                env.AmbientLightColor = C("#9DAEE6");
                env.BackgroundColor = C("#0A0E18");
                break;
            case "dusk":
                sun.LightEnergy = 0.5f;
                sun.LightColor = C("#FFB877");
                env.AmbientLightEnergy = 0.42f;
                env.AmbientLightColor = C("#D7C3E6");
                break;
            case "rain":
                sun.LightEnergy = 0.42f;
                sun.LightColor = C("#C9D3DC");
                env.AmbientLightEnergy = 0.8f;
                env.AmbientLightColor = C("#B9C6D2");
                env.AdjustmentSaturation = 0.92f;
                break;
            default:
                env.AdjustmentSaturation = 1.12f;
                break;
        }
    }

    private static void Glow(Node3D parent, Mesh mesh, Vector3 centre, Vector3 size, Color colour, float energy = 2.5f, float alpha = 1f)
    {
        var m = new StandardMaterial3D
        {
            AlbedoColor = new Color(colour, alpha), EmissionEnabled = true, Emission = colour, EmissionEnergyMultiplier = energy,
            ShadingMode = BaseMaterial3D.ShadingModeEnum.Unshaded,
            Transparency = alpha < 1 ? BaseMaterial3D.TransparencyEnum.Alpha : BaseMaterial3D.TransparencyEnum.Disabled,
            CullMode = BaseMaterial3D.CullModeEnum.Disabled,
        };
        if (alpha < 1)
        {
            m.BlendMode = BaseMaterial3D.BlendModeEnum.Add;
        }

        parent.AddChild(new MeshInstance3D { Mesh = mesh, Position = centre, Scale = size, MaterialOverride = m, CastShadow = GeometryInstance3D.ShadowCastingSetting.Off });
    }

    /// <summary>A pitched roof over a rectangle: ridge along X.</summary>
    private static void Pitched(MeshBatch b, float x0, float x1, float z0, float z1, float eave, float ridge, Color colour, Color? end = null)
    {
        var zm = (z0 + z1) / 2;
        Vector3 a = new(x0, eave, z0), bb = new(x1, eave, z0), c = new(x1, eave, z1), d = new(x0, eave, z1), r0 = new(x0, ridge, zm), r1 = new(x1, ridge, zm);
        Quad(b, d, c, r1, r0, colour);
        Quad(b, bb, a, r0, r1, colour.Darkened(0.08f));
        b.Tri(a, d, r0, end ?? colour.Darkened(0.04f));
        b.Tri(c, bb, r1, end ?? colour.Darkened(0.04f));
    }

    private static void Quad(MeshBatch b, Vector3 p0, Vector3 p1, Vector3 p2, Vector3 p3, Color colour)
    {
        b.Tri(p0, p1, p2, colour);
        b.Tri(p0, p2, p3, colour);
        b.Tri(p2, p1, p0, colour);
        b.Tri(p3, p2, p0, colour);
    }

    /// <summary>A pyramid (marquee peak) over a rectangle.</summary>
    private static void Peak(MeshBatch b, float x0, float x1, float z0, float z1, float eave, float top, Color colour, Color? alt = null)
    {
        Vector3 a = new(x0, eave, z0), bb = new(x1, eave, z0), c = new(x1, eave, z1), d = new(x0, eave, z1), t = new((x0 + x1) / 2, top, (z0 + z1) / 2);
        var o = alt ?? colour.Darkened(0.07f);
        foreach (var (p, q, col) in new[] { (d, c, colour), (c, bb, o), (bb, a, colour.Darkened(0.1f)), (a, d, o) })
        {
            b.Tri(p, q, t, col);
            b.Tri(q, p, t, col);
        }
    }

    private static void Truss(MeshBatch b, Vector3 from, Vector3 to, float w, Color colour)
    {
        var up = Mathf.Abs((to - from).Normalized().Y) > 0.9f;
        var o1 = up ? new Vector3(w, 0, 0) : new Vector3(0, w, 0);
        var o2 = new Vector3(0, 0, w);
        foreach (var o in new[] { o1 + o2, o1 - o2, -o1 + o2, -o1 - o2 })
        {
            b.Rod(from + (o / 2), to + (o / 2), 0.045f, colour, true);
        }

        var len = (to - from).Length();
        var steps = Math.Max(1, (int)(len / (w * 1.4f)));
        for (var i = 0; i < steps; i++)
        {
            var p = from.Lerp(to, i / (float)steps);
            var q = from.Lerp(to, (i + 1) / (float)steps);
            b.Rod(p + ((o1 + o2) / 2), q + ((-o1 + o2) / 2), 0.03f, colour, true);
            b.Rod(p + ((o1 - o2) / 2), q + ((-o1 - o2) / 2), 0.03f, colour, true);
        }
    }

    private static float HeadY(PersonLook l) => Person3D.V((43.2f * l.Scale) + (6.45f * l.HeadScale) + 1);

    private void Hat(Person3D p, Color colour, bool cowboy = false)
    {
        var b = new MeshBatch();
        var r = Person3D.W(7.5f * p.Look.HeadScale);
        var y = HeadY(p.Look);
        b.Cylinder(new Vector3(0, y + (r * 0.62f), 0), r * (cowboy ? 1.75f : 1.45f), 0.03f, r * (cowboy ? 1.75f : 1.45f), colour.Darkened(0.06f));

        b.AddTo(p);
    }

    private static void ArmsUp(Person3D p, double t, int kind)
    {
        var s = (float)Math.Sin(t);
        switch (kind)
        {
            case 0: p.SetArms(new Vector3(-2.75f + (s * 0.2f), 0, -0.25f), new Vector3(-2.75f - (s * 0.2f), 0, 0.25f)); break;
            case 1: p.SetArms(new Vector3(-0.1f, 0, 0.06f), new Vector3(-2.6f + (s * 0.25f), 0, 0.15f)); break;
            case 2: p.SetArms(new Vector3(-1.5f, 0, -0.5f), new Vector3(-1.5f, 0, 0.5f)); break;
        }
    }

    private void Crowd(Node3D parent, double cx, double cy, Func<double, double, double> density, int tries, double halfW, double depth, double faceDeg, bool cheer, bool rain = false)
    {
        // Local: u across, v away from the stage (stage faces +v). World = centre + u*(1,-1)/√2 + v*(1,1)/√2.
        var placed = new List<(double X, double Y)>();
        for (var i = 0; i < tries; i++)
        {
            var u = (R() - 0.5) * 2 * halfW;
            var v = R() * depth;
            if (R() > density(u, v))
            {
                continue;
            }

            var x = cx + ((u + v) * 0.7071);
            var y = cy + ((v - u) * 0.7071);
            if (placed.Any(q => Math.Abs(q.X - x) + Math.Abs(q.Y - y) < 0.72))
            {
                continue;
            }

            placed.Add((x, y));
            var look = Goer(1000 + i, rain);
            var p = Put(parent, look, x, y, faceDeg + ((R() - 0.5) * 30));
            if (cheer && R() < 0.55)
            {
                ArmsUp(p, R() * 6, Pick(0, 0, 1, 1, 2));
            }

            if (R() < 0.13)
            {
                Hat(p, C(Pick("#F7F3E8", "#F2517A", "#F6B93B", "#36B5A2", "#D9C29A", "#58C4DD")), R() < 0.3);
            }

            _crowd.Add(p);
        }
    }

    private void Ground()
    {
        var b = new MeshBatch();
        b.Box(new Vector3(130, -0.26f, 130), new Vector3(620, 0.5f, 620), C("#5E9C4E"));

        // Mown stripes, diagonal to the grid so they run across the screen.
        var g = new Node3D { Rotation = new Vector3(0, Mathf.Pi / 4, 0), Position = Kit3D.At(62, 62) };
        _site.AddChild(g);
        var s = new MeshBatch();
        for (var i = -40; i < 40; i += 2)
        {
            s.Box(new Vector3(0, 0.002f, i * 4f), new Vector3(420, 0.012f, 4f), C("#66A655"));
        }

        s.AddTo(g, shadow: false);

        // Worn grass in front of the stage, and trackway paths.
        var w = new MeshBatch();
        w.Cylinder(new Vector3(0, 0.01f, 16), 20, 0.02f, 12, C("#86A85A"));
        w.Cylinder(new Vector3(0, 0.014f, 10), 13, 0.02f, 5.5f, C("#9AA864"));
        w.AddTo(Holder(_dry, 36, 36), shadow: false);
        var mud = new MeshBatch();
        mud.Cylinder(new Vector3(0, 0.01f, 16), 20, 0.02f, 12, C("#7A6A46"));
        mud.Cylinder(new Vector3(0, 0.014f, 10), 13, 0.02f, 5.5f, C("#6A5638"));
        mud.AddTo(Holder(_wet, 36, 36), shadow: false);

        // Trackway: the main avenue from the entrance past the food row, and a spur to the campsite.
        Track(b, 100, 96, 100, 44, 3.2f);
        Track(b, 100, 70, 58, 78, 2.6f);
        Track(b, 58, 78, 30, 84, 2.6f);
        Track(b, 100, 58, 84, 58, 2.6f);
        b.AddTo(_site, shadow: false);
    }

    private static void Track(MeshBatch b, double x0, double y0, double x1, double y1, float width)
    {
        var a = Kit3D.At(x0, y0, 0.02f);
        var c = Kit3D.At(x1, y1, 0.02f);
        var len = (c - a).Length();
        var d = (c - a).Normalized();
        var basis = new Basis(Vector3.Up, Mathf.Atan2(d.X, d.Z));
        var n = (int)(len / 1.2f);
        for (var i = 0; i <= n; i++)
        {
            var p = a.Lerp(c, i / (float)Math.Max(1, n));
            b.Box(p, new Vector3(width * 2, 0.03f, 1.12f), i % 2 == 0 ? C("#C4C8BE") : C("#B5BAB0"), basis);
        }
    }

    private void MainStage()
    {
        var h = Holder(_site, 36, 36);
        var b = new MeshBatch();
        var steel = C("#8E99A3");
        var black = C("#1B1E24");
        float deckH = Hm(1.9), roofH = Hm(11.5), W = 19, D = 13;

        // Deck with a skirt and steps.
        b.Box(new Vector3(0, deckH / 2, -D / 2), new Vector3(W * 2, deckH, D), C("#2A2E36"));
        b.Box(new Vector3(0, deckH + 0.02f, -D / 2), new Vector3((W * 2) - 0.4f, 0.04f, D - 0.4f), C("#3A3F49"));
        b.Box(new Vector3(0, deckH / 2, 0.03f), new Vector3(W * 2, deckH * 0.9f, 0.06f), C("#16181C"));

        // Four towers, roof grid and a curved-look roof skin.
        foreach (var x in new[] { -W + 0.6f, W - 0.6f })
        {
            foreach (var z in new[] { -D + 0.6f, -0.6f })
            {
                Truss(b, new Vector3(x, 0, z), new Vector3(x, roofH, z), 0.7f, steel);
            }
        }

    private void Festoon(double x0, double y0, double x1, double y1, float side)
    {
        var a = Kit3D.At(x0, y0);
        var c = Kit3D.At(x1, y1);
        var dir = (c - a).Normalized();
        var n = new Vector3(-dir.Z, 0, dir.X) * side;
        var len = (c - a).Length();
        var b = new MeshBatch();
        var poles = (int)(len / 7);
        var cols = new[] { "#FFE9B0", "#F2517A", "#58C4DD", "#F6B93B", "#9BD770" };
        for (var i = 0; i <= poles; i++)
        {
            var p = a + (dir * (i * len / poles)) + n;
            b.Rod(p, p + new Vector3(0, Hm(4), 0), 0.05f, C("#6B5237"), true);
            if (i == poles)
            {
                break;
            }

            var q = a + (dir * ((i + 1) * len / poles)) + n;
            for (var k = 1; k < 8; k++)
            {
                var f = k / 8f;
                var at = p.Lerp(q, f) + new Vector3(0, Hm(4) - (Mathf.Sin(f * Mathf.Pi) * 0.5f), 0);
                Glow(_site, Kit3D.SmallSphere, at, Vector3.One * 0.2f, C(cols[(i + k) % 5]), 2.4f);
            }

            _nightOnly.AddChild(new OmniLight3D { Position = p.Lerp(q, 0.5f) + new Vector3(0, Hm(3.4), 0), LightColor = C("#FFD9A0"), LightEnergy = 1.5f, OmniRange = 9 });
        }

        b.AddTo(_site);
    }

    private void Marquee(MeshBatch b, float w, float d, float eave, float top, Color roof, Color? alt = null, bool back = true, bool sides = true)
    {
        Peak(b, (-w / 2) - 0.2f, (w / 2) + 0.2f, (-d / 2) - 0.2f, (d / 2) + 0.2f, eave, top, roof, alt);

    }

    private void MarqueeFrame(MeshBatch b, float w, float d, float eave, Color roof, bool back = true, bool sides = true)
    {
        foreach (var x in new[] { -w / 2, w / 2 })
        {
            foreach (var z in new[] { -d / 2, d / 2 })
            {
                b.Rod(new Vector3(x, 0, z), new Vector3(x, eave, z), 0.06f, C("#E3E8EC"), true);
            }
        }

        b.Box(new Vector3(0, eave - 0.14f, d / 2 + 0.2f), new Vector3(w + 0.4f, 0.3f, 0.04f), roof.Darkened(0.04f));
        b.Box(new Vector3(0, eave - 0.14f, -d / 2 - 0.2f), new Vector3(w + 0.4f, 0.3f, 0.04f), roof.Darkened(0.04f));
        b.Box(new Vector3(w / 2 + 0.2f, eave - 0.14f, 0), new Vector3(0.04f, 0.3f, d + 0.4f), roof.Darkened(0.08f));
        b.Box(new Vector3(-w / 2 - 0.2f, eave - 0.14f, 0), new Vector3(0.04f, 0.3f, d + 0.4f), roof.Darkened(0.08f));
        if (back)
        {
            b.Box(new Vector3(0, eave / 2, -d / 2), new Vector3(w, eave, 0.05f), roof.Darkened(0.06f));
        }

        if (sides)
        {
            b.Box(new Vector3(-w / 2, eave / 2, -d / 4), new Vector3(0.05f, eave, d / 2), roof.Darkened(0.1f));
        }
    }

    private void Tent(MeshBatch b, Vector3 c, float yaw, int kind, Color col, bool wet)
    {
        var basis = new Basis(Vector3.Up, yaw);
        var trim = col.Darkened(0.18f);
        switch (kind)
        {
            case 0: // dome
                b.Ball(c + new Vector3(0, 0.05f, 0), new Vector3(3.0f, 2.3f, 3.0f), basis, col);
                b.Ball(c + (basis * new Vector3(0, 0.0f, 1.0f)), new Vector3(1.5f, 1.5f, 1.5f), basis, trim);
                break;
            case 1: // tunnel
                b.Ball(c + new Vector3(0, 0.0f, 0), new Vector3(2.8f, 2.4f, 5.4f), basis, col);
                b.Ball(c + (basis * new Vector3(0, 0, 2.0f)), new Vector3(2.2f, 1.9f, 1.8f), basis, trim);
                break;
            case 2: // bell tent
                b.Cylinder(c + new Vector3(0, 0.45f, 0), 2.1f, 0.9f, 2.1f, C("#EFE6D2"));
                b.Cylinder(c + new Vector3(0, 0.9f + 0.9f, 0), 2.25f, 1.8f, 2.25f, C("#F7F0DE"), null, true);
                b.Box(c + (basis * new Vector3(0, 0.45f, 2.06f)), new Vector3(0.9f, 0.9f, 0.1f), C("#B9A27A"), basis);
                break;
            default: // ridge
                var x0 = -1.2f;
                var x1 = 1.2f;
                var pts = new[] { new Vector3(x0, 0, -1.6f), new Vector3(x1, 0, -1.6f), new Vector3(x1, 0, 1.6f), new Vector3(x0, 0, 1.6f), new Vector3(0, 1.25f, -1.6f), new Vector3(0, 1.25f, 1.6f) }.Select(p => c + (basis * p)).ToArray();
                Quad(b, pts[0], pts[3], pts[5], pts[4], col);
                Quad(b, pts[2], pts[1], pts[4], pts[5], col.Darkened(0.08f));
                b.Tri(pts[3], pts[2], pts[5], trim);
                b.Tri(pts[1], pts[0], pts[4], trim);
                break;
        }

        _ = wet;
    }

                    var dy = p.Position.Z - cy;
                    var pu = (dx - dy) * 0.7071;
                    var pv = (dx + dy) * 0.7071;
                    if (pu >= u && pu < u + cell && pv >= v && pv < v + cell)
                    {
                        count++;
                    }
                }

                if (count == 0)
                {
                    continue;
                }

                // Each mock-up figure stands for several people: scale up so the front reads as a real pinch point.
                var perM2 = count * 0.5 * (v < 5 && Math.Abs(u + 4) < 9 ? 1.4 : 1);
                var col = perM2 >= 4.9 ? C("#D94A4A") : perM2 >= 3.9 ? C("#F2A33A") : C("#2FA05A");
                var mu = u + (cell / 2);
                var mv = v + (cell / 2);
                g.Box(Kit3D.At(cx + ((mu + mv) * 0.7071), cy + ((mv - mu) * 0.7071), Hm(2.4)), new Vector3((float)cell - 0.12f, 0.02f, (float)cell - 0.12f), new Color(col, perM2 >= 4.9 ? 0.6f : perM2 >= 3.9 ? 0.48f : 0.3f), new Basis(Vector3.Up, Mathf.Pi / 4));
            }
        }

        var (wx, wy) = P(-4, 3 + 5.6);
        Sign(_density, "CRUSH RISK  4.8 per m²", Kit3D.At(wx, wy, Hm(4.6)), C("#FFFFFF"), 0.016f, C("#B3261E"), true);
        var (ox, oy) = P(17, 14);
        Sign(_density, "Busy  3.1 per m²", Kit3D.At(ox, oy, Hm(4.2)), C("#FFFFFF"), 0.012f, C("#B56A00"), true);
        var (gx, gy) = P(-14, 33);
        Sign(_density, "Comfortable  1.4 per m²", Kit3D.At(gx, gy, Hm(4.2)), C("#FFFFFF"), 0.012f, C("#1E7A42"), true);
        var mi = g.AddTo(_density, alpha: true, shadow: false);
        if (mi is not null)
        {
            mi.MaterialOverride = new StandardMaterial3D { VertexColorUseAsAlbedo = true, ShadingMode = BaseMaterial3D.ShadingModeEnum.Unshaded, Transparency = BaseMaterial3D.TransparencyEnum.Alpha, NoDepthTest = false };
        }
    }
}

internal static class NodeExt
{
    public static T With<T>(this T node, Action<T> act)
        where T : Node
    {
        act(node);
        return node;
    }
}
